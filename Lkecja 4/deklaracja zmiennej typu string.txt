string ZmienionyTekst = Reeplace("Nie kupuje kubkok","k","x");

string Reeplace(string TekstDoZmiany,string Co, string NaCo)
{

    string TekstPoZmianie = "";
    bool czyJestCo = TekstDoZmiany.Contains(Co);
    //Sprawdfzam czy jest co zmieniać
    if (czyJestCo)
    {
        while (czyJestCo)
        {
            int pos = TekstDoZmiany.IndexOf(Co);
            if (pos == 0)//Wyszykiwany tekst znajduje się na pierwszej pozycji
            {
                TekstPoZmianie = NaCo + TekstDoZmiany.Substring(pos + 1);
            }
            else if (pos > 0 && pos < TekstDoZmiany.Length - NaCo.Length)//Wyszukiwany tekst znajduje się na środkowej pozycji
            {
                TekstPoZmianie = TekstDoZmiany.Substring(0, pos) + NaCo + TekstDoZmiany.Substring(pos + Co.Length) ;
            }
            else if(pos > 0 && pos == TekstDoZmiany.Length - NaCo.Length)//Wyszukiwany tekst znajduje się na ostatniej pozycji
            {
                TekstPoZmianie = TekstDoZmiany.Substring(0, pos) + NaCo;
            }

            TekstDoZmiany = TekstPoZmianie;
            czyJestCo = TekstDoZmiany.Contains(Co);
        }

    }





    return TekstPoZmianie;
}

Console.WriteLine(ZmienionyTekst);
